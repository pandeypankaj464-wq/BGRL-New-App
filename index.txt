const K=Object.keys(M).sort().reverse();return cash?tbl(['Period','Inflow (received)','Outflow (owners)','Expenses','Net cash'],K.map(k=>[k,fmt(M[k].rc),fmt(M[k].pd),fmt(M[k].ex),fmt(M[k].rc-M[k].pd-M[k].ex)])):tbl(['Period','Bookings','Freight','Brokerage','Received','Paid to owners','Expenses','Net income'],K.map(k=>[k,M[k].bk,fmt(M[k].fr),fmt(M[k].br),fmt(M[k].rc),fmt(M[k].pd),fmt(M[k].ex),fmt(M[k].br-M[k].ex)]))}

function parseCSV(x){const R=[];let r=[],c='',q=0;for(let i=0;i<x.length;i++){const h=x[i];if(q){if(h=='"'){if(x[i+1]=='"'){c+='"';i++}else q=0}else c+=h}else if(h=='"')q=1;else if(h==','){r.push(c);c=''}else if(h=='\n'||h=='\r'){if(h=='\r'&&x[i+1]=='\n')i++;r.push(c);c='';R.push(r);r=[]}else c+=h}if(c||r.length){r.push(c);R.push(r)}return R.filter(z=>z.join(''))}

if(!H.includes(key))return alert('CSV must have a "'+key+'" column. Nothing imported.');
const rows=R.map(c=>{const o={id:uid(),ts:Date.now()};H.forEach((h,i)=>{if(keys.includes(h))o[h]=SC[s].f.find(f=>f.k==h).ty=='number'?n(c[i]):(c[i]||'')});return o}).filter(o=>o[key]);
